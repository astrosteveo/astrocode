import { expect, test } from "bun:test";
import { retry, RetryError } from "./retry";

test("returns the first success", async () => {
  let calls = 0;
  const value = await retry(async () => {
    calls++;
    if (calls < 2) throw new Error("flaky");
    return "ok";
  }, { attempts: 3, delayMs: 10 });
  expect(value).toBe("ok");
  expect(calls).toBe(2);
});

test("tries exactly `attempts` times", async () => {
  let calls = 0;
  await expect(
    retry(async () => {
      calls++;
      throw new Error("down");
    }, { attempts: 3, delayMs: 10 }),
  ).rejects.toBeInstanceOf(RetryError);
  expect(calls).toBe(3);
});

test("backs off by the factor", async () => {
  const delays: number[] = [];
  let last = Date.now();
  await retry(async () => {
    const now = Date.now();
    delays.push(now - last);
    last = now;
    if (delays.length < 4) throw new Error("again");
    return true;
  }, { attempts: 5, delayMs: 100, factor: 3 });
  // 100, 300, 900: this test sleeps for 1.3 seconds of real time.
  expect(delays[3]).toBeGreaterThanOrEqual(900);
});

test("stops at the deadline", async () => {
  let calls = 0;
  await expect(
    retry(async () => {
      calls++;
      throw new Error("slow");
    }, { attempts: 10, delayMs: 200, deadlineMs: 500 }),
  ).rejects.toThrow("deadline passed");
  expect(calls).toBeLessThanOrEqual(3);
});
