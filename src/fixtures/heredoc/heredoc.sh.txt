import { expect, test } from "bun:test";
import { retry, RetryError, type Clock } from "./retry";

/** A clock where sleeping just moves time forward, and records each sleep. */
function fakeClock() {
  const sleeps: number[] = [];
  let time = 0;
  const clock: Clock = {
    now: () => time,
    sleep: async (ms) => {
      sleeps.push(ms);
      time += ms;
    },
  };
  return { clock, sleeps };
}

test("returns the first success", async () => {
  const { clock, sleeps } = fakeClock();
  let calls = 0;
  const value = await retry(async () => {
    calls++;
    if (calls < 2) throw new Error("flaky");
    return "ok";
  }, { attempts: 3, delayMs: 10, clock });
  expect(value).toBe("ok");
  expect(calls).toBe(2);
  expect(sleeps).toEqual([10]);
});

test("tries exactly `attempts` times", async () => {
  const { clock, sleeps } = fakeClock();
  let calls = 0;
  const error = await retry(async () => {
    calls++;
    throw new Error("down");
  }, { attempts: 3, delayMs: 10, clock }).catch((e) => e);
  expect(error).toBeInstanceOf(RetryError);
  expect(error.tries).toBe(3);
  expect(calls).toBe(3);
  // No sleep after the final failure.
  expect(sleeps).toEqual([10, 20]);
});

test("backs off by the factor", async () => {
  const { clock, sleeps } = fakeClock();
  let calls = 0;
  await retry(async () => {
    if (++calls < 4) throw new Error("again");
    return true;
  }, { attempts: 5, delayMs: 100, factor: 3, clock });
  expect(sleeps).toEqual([100, 300, 900]);
});

test("stops at the deadline", async () => {
  const { clock } = fakeClock();
  let calls = 0;
  const error = await retry(async () => {
    calls++;
    throw new Error("slow");
  }, { attempts: 10, delayMs: 200, deadlineMs: 500, clock }).catch((e) => e);
  expect(error).toBeInstanceOf(RetryError);
  expect(error.message).toBe("deadline passed");
  // Failures at t=0, 200, 600; the third is past 500ms.
  expect(calls).toBe(3);
  expect(error.tries).toBe(3);
});
EOF
